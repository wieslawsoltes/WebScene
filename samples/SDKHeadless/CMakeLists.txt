cmake_minimum_required(VERSION 3.28)
project(WebSceneHeadless LANGUAGES CXX)
find_package(WebScene REQUIRED CONFIG)
include(CTest)
add_executable(native_document main.cpp)
target_link_libraries(native_document PRIVATE WebScene::NativeWeb)
webscene_compile_html(native_document index.html MODULE sample.ui)
add_test(NAME sdk_native_document COMMAND native_document)
option(WEBSCENE_HEADLESS_RUNTIME_TESTS "Build reusable-runtime ownership contracts" OFF)
if(WEBSCENE_HEADLESS_RUNTIME_TESTS)
  add_executable(runtime_contracts runtime_contracts.cpp)
  target_link_libraries(runtime_contracts PRIVATE WebScene::Runtime)
  webscene_stage_runtime(runtime_contracts)
  add_test(NAME sdk_runtime_contracts COMMAND runtime_contracts)
  set_tests_properties(sdk_runtime_contracts PROPERTIES TIMEOUT 45)
endif()
