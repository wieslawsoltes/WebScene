cmake_minimum_required(VERSION 3.28)
project(WebSceneHeadlessConsumer LANGUAGES CXX)
include(CTest)
find_package(WebScene REQUIRED CONFIG COMPONENTS NativeWeb SharedCSS Compiler)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HEADLESS_TEST_WEBGPU "Exercise the installed Vulkan SDK" ON)
if(HEADLESS_TEST_WEBGPU)
  find_package(WebScene REQUIRED CONFIG COMPONENTS WebGPU)
  if(NOT TARGET WebScene::WebGPU)
    message(FATAL_ERROR "The headless GPU contract requires the real WebGPU component")
  endif()
  add_executable(headless_gpu headless_gpu.cpp)
  target_link_libraries(headless_gpu PRIVATE WebScene::WebGPU)
  add_test(NAME headless_gpu COMMAND headless_gpu)
  set_tests_properties(headless_gpu PROPERTIES TIMEOUT 180)
endif()
