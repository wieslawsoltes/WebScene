function(webscene_compile_html target input)
 set(output "${CMAKE_CURRENT_BINARY_DIR}/${target}_ui.hpp")
 add_custom_command(OUTPUT "${output}" COMMAND webscene-uic "${input}" "${output}" DEPENDS webscene-uic "${input}" DEPFILE "${output}.d" VERBATIM)
 target_sources(${target} PRIVATE "${output}")
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
endfunction()
function(webscene_compile_html_module target input module_name)
 set(output "${CMAKE_CURRENT_BINARY_DIR}/${target}_ui.cppm")
 add_custom_command(OUTPUT "${output}" COMMAND webscene-uic "${input}" "${output}" --module "${module_name}" ${ARGN} DEPENDS webscene-uic "${input}" DEPFILE "${output}.d" VERBATIM)
 target_sources(${target} PUBLIC FILE_SET generated_ui TYPE CXX_MODULES BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}" FILES "${output}")
endfunction()
function(webscene_prepare_css_module target input module_name)
 set(output "${CMAKE_CURRENT_BINARY_DIR}/${target}_css.cppm")
 add_custom_command(OUTPUT "${output}" COMMAND webscene-uic --prepare-css "${input}" "${output}" --module "${module_name}" DEPENDS webscene-uic "${input}" VERBATIM)
 target_sources(${target} PUBLIC FILE_SET generated_css TYPE CXX_MODULES BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}" FILES "${output}")
endfunction()
